volatile uint32_t timerz = 0;

{

    pin_mod('B', 5, 1);
    uint8_t sec_info = 0;

    _delay_ms(1000);
    lcd_yaz(0, "     -SEC-");

        sec_info = pinR('D', 4);
        if (timern == 0) {
            pinW('D', 6, 1);

            _delay_ms(100);
            pinW('D', 6, 1);
            _delay_ms(100);
            pinW('D', 6, 0);
            _delay_ms(100);
        }

            lcd_yaz(1, "-Movement-");
            pinW('D', 6, 0);
            _delay_ms(150);
            pinW('D', 6, 1);
             timerz = !timerz;

        }
        else {
            pinW('D', 6, 1);
           
            lcd_yaz(1, "-Sleep-");
        }       
        
    }

}
