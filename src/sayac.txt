#include "gpio.h"
#include "lcd.h"
#include "interrupt.h" // Senin kesme kütüphanen
#include "uart.h"
#include <util/delay.h>

volatile uint8_t i = 0;  // On/Off durumu
volatile uint32_t s = 0; // Sayaç

// INT0 (D2) Kesmesi: On/Off
ISR(INT0_vect)
{
    i = !i;
}

// INT1 (D3) Kesmesi: Reset
ISR(INT1_vect)
{
    s = 0;
}

int main(void)
{
    // 1. Başlatmalar
    lcd_basla();
    lcd_temizle();
    uart_basla();

    // 2. Buton Pinleri
    pin_mod('D', 2, 0); // INT0
    pinW('D', 2, 1);    // Pull-up aç
    pin_mod('D', 3, 0); // INT1
    pinW('D', 3, 1);    // Pull-up aç
    pin_mod('D', 6, 1);
    pin_mod('D', 5, 1);

    // 3. Kesmeleri Başlat (Senin kütüphanenle)
    kesme_basla(KESME_INT0, TETIK_DUSEN);
    kesme_basla(KESME_INT1, TETIK_DUSEN);
    sei();
    pinW('D', 5, 1);
    _delay_ms(150);
    pinW('D', 5, 0);
    // 4. Başlangıç
    lcd_yaz(0, "SAYAC:");
    lcd_degerW(1, s);

    while (1)
    {
        if (i == 0)
        {
            pinW('D', 6, 1);
            lcd_degerW(1, s);
            s++;
            pinW('D', 5, 1);
            _delay_ms(100);

            // Ekrana yaz

            // 1 saniye bekle (ama kesmeleri bloke etme)
            for (uint8_t j = 0; j < 90; j++)
            {
                pinW('D', 5, 0);
                _delay_us(9810);
            }
        }
        else
        {
            // Durdurulduysa sadece bekle
            pinW('D', 6, 0);
        }

        // Debug
        uart_degerW(i);
        uart_yaz(" | ");
        uart_degerW(s);
        uart_yaz("\r\n");
    }
}
